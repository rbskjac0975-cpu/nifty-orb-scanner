    <script>
        // Interactive Canvas Charting System
        const canvas = document.getElementById('mainChart');
        const ctx = canvas.getContext('2d');

        function resizeCanvas() {
            canvas.width = canvas.parentElement.clientWidth;
            canvas.height = canvas.parentElement.clientHeight;
            drawChart();
        }
        window.addEventListener('resize', resizeCanvas);

        // Dummy Candle Data for 1-Minute Chart
        let candles = [
            { open: 2440, high: 2448, low: 2438, close: 2445 },
            { open: 2445, high: 2452, low: 2442, close: 2450 },
            { open: 2450, high: 2458, low: 2448, close: 2456 },
            { open: 2456, high: 2462, low: 2453, close: 2460 },
            { open: 2460, high: 2468, low: 2458, close: 2465 }
        ];

        function drawChart() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            const padding = 40;
            const width = canvas.width - padding * 2;
            const height = canvas.height - padding * 2;
            
            const candleWidth = width / candles.length - 15;
            
            // Find High/Low
            let minPrice = Math.min(...candles.map(c => c.low));
            let maxPrice = Math.max(...candles.map(c => c.high));

            candles.forEach((c, index) => {
                const x = padding + index * (candleWidth + 15) + 20;
                const openY = canvas.height - padding - ((c.open - minPrice) / (maxPrice - minPrice)) * height;
                const closeY = canvas.height - padding - ((c.close - minPrice) / (maxPrice - minPrice)) * height;
                const highY = canvas.height - padding - ((c.high - minPrice) / (maxPrice - minPrice)) * height;
                const lowY = canvas.height - padding - ((c.low - minPrice) / (maxPrice - minPrice)) * height;

                const isBull = c.close >= c.open;
                ctx.strokeStyle = isBull ? '#22c55e' : '#ef4444';
                ctx.fillStyle = isBull ? '#22c55e' : '#ef4444';

                // Wick
                ctx.beginPath();
                ctx.moveTo(x + candleWidth / 2, highY);
                ctx.lineTo(x + candleWidth / 2, lowY);
                ctx.stroke();

                // Body
                ctx.fillRect(x, Math.min(openY, closeY), candleWidth, Math.max(Math.abs(closeY - openY), 2));
            });
        }

        // Initialize Mock Scanner Signals
        const stocks = [
            { symbol: 'RELIANCE', price: '₹2,465.00', change: '+1.2%', type: 'BULL' },
            { symbol: 'TATAMOTORS', price: '₹985.50', change: '+2.4%', type: 'BULL' },
            { symbol: 'INFY', price: '₹1,510.20', change: '-0.8%', type: 'BEAR' },
            { symbol: 'HDFCBANK', price: '₹1,640.00', change: '+0.5%', type: 'BULL' },
            { symbol: 'ICICIBANK', price: '₹1,120.30', change: '+1.1%', type: 'BULL' }
        ];

        function renderScanner() {
            const list = document.getElementById('scannerList');
            list.innerHTML = stocks.map(s => `
                <div class="stock-item ${s.symbol === 'RELIANCE' ? 'active' : ''}">
                    <div>
                        <div class="stock-symbol">${s.symbol}</div>
                        <span class="badge ${s.type === 'BULL' ? 'badge-bull' : 'badge-bear'}">1M ${s.type} ORB</span>
                    </div>
                    <div class="stock-price">
                        <div>${s.price}</div>
                        <div style="color: ${s.type === 'BULL' ? 'var(--accent-green)' : 'var(--accent-red)'}">${s.change}</div>
                    </div>
                </div>
            `).join('');
        }

        function triggerTrade(action) {
            alert(action + " Order Sent to Broker API Interface!");
        }

        // Initial Setup
        setTimeout(() => {
            resizeCanvas();
            renderScanner();
        }, 100);
    </script>
